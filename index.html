function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "getNecessidades") { var sheet = ss.getSheetByName("Gestao_Necessidades"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); var result = []; // Ignora o cabeçalho (linha 0) for (var i = 1; i < data.length; i++) { result.push({ linhaExcel: i + 1, idPedido: data[i][0], // Coluna A dataRegisto: data[i][1], // Coluna B tipoValência: data[i][2], // Coluna C diaSemana: data[i][3], // Coluna D tempoLetivo: data[i][4], // Coluna E entidadeAlvo: data[i][5], // Coluna F localizacao: data[i][6], // Coluna G prioridade: data[i][7], // Coluna H descricaoObjetivos: data[i][8],// Coluna I estado: data[i][9], // Coluna J profAssumiu: data[i][10], // Coluna K dataAssumido: data[i][11], // Coluna L urlDocumento: data[i][12] // Coluna M }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; var payload = data.payload; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Gestao_Necessidades"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Folha Gestao_Necessidades não encontrada." })) .setMimeType(ContentService.MimeType.JSON); } if (action === "novaNecessidade") { var fileUrl = ""; if (payload.fileData && payload.fileName) { var folder = DriveApp.getRootFolder(); var contentType = payload.mimeType || "application/octet-stream"; var bytes = Utilities.base64Decode(payload.fileData.split(',')[1]); var blob = Utilities.newBlob(bytes, contentType, payload.fileName); var file = folder.createFile(blob); fileUrl = file.getUrl(); } var idPedido = "REQ-" + new Date().getTime(); // Guarda estritamente pela ordem A -> M: // A: ID, B: DataRegisto, C: TipoValência, D: Dia, E: Tempo, F: Entidade, G: Local, H: Prioridade, I: Descrição, J: Estado, K: Prof, L: DataAssumido, M: URL sheet.appendRow([ idPedido, payload.dataRegisto, payload.tipoValência, payload.diaSemana, payload.tempoLetivo, payload.entidadeAlvo, payload.localizacao, payload.prioridade, payload.descricaoObjetivos, "Pendente", // Coluna J (Estado) "", // Coluna K (Professor_Alocado) "", // Coluna L (Data Assumido) fileUrl // Coluna M (URL) ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", idPedido: idPedido })) .setMimeType(ContentService.MimeType.JSON); } if (action === "assumirApoio") { var dataNecessidades = sheet.getDataRange().getValues(); for (var i = 1; i < dataNecessidades.length; i++) { if (String(dataNecessidades[i][0]) === String(payload.idPedido)) { var linha = i + 1; // Coluna J (10) -> Estado = "Em Execução" sheet.getRange(linha, 10).setValue("Em Execução"); // Coluna K (11) -> Professor_Alocado = Nome do Professor sheet.getRange(linha, 11).setValue(payload.nomeProfessor); // Coluna L (12) -> Data Assumido = Data e hora atual sheet.getRange(linha, 12).setValue(payload.dataAssumido); break; } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
